17def runner_for(row, library_runner: str | None, node_id: str) -> str: 

18 """item node override > item-wide override > library (step > node > chain) > local.""" 

19 node = (store.node_overrides_of(row).get(node_id) or {}).get("runner") 

20 raw = row["agent_overrides"] 

21 item = (json.loads(raw) if raw else {}).get("runner") 

22 return node or item or library_runner or LOCAL

25def library_runners(chain) -> set[str]: 

26 """Every machine (or `any`) the library places a step on; `chain` is a `ResolvedChain` or a 

27 `MaterializedChain`.""" 

28 resolved = chain.chain if hasattr(chain, "target") else chain 

29 return {t.runner for n in resolved.nodes for t in n.tasks() if t.runner} - {LOCAL}

32def repos_label(repos: Sequence[RepoEntry]) -> str: 

33 return ", ".join(r.name or Path(r.path).name for r in repos) or "this item"

36def allows(worker: registry.Worker, repos: Sequence[RepoEntry]) -> bool: 

37 if worker.repos is None: 

38 return True 

39 return all(any(registry.repo_matches(r, n) for n in worker.repos) for r in repos)

42def refusal( 

43 runner: str, *, workers: registry.WorkersFile, repos: Sequence[RepoEntry] 

44) -> str | None: 

45 if runner == LOCAL: 

46 return None 

47 if runner == ANY: 

48 if any(allows(w, repos) for w in workers.workers.values()): 

49 return None 

50 return ( 

51 f"runner 'any' finds no machine in workers.yaml that may work on {repos_label(repos)}" 

52 ) 

53 worker = workers.workers.get(runner) 

54 if worker is None: 

55 return f"runner {runner!r} is not a machine in workers.yaml" 

56 for r in repos: 

57 if not allows(worker, [r]): 

58 return f"machine {runner!r} is not allowed for repo {(r.name or Path(r.path).name)!r}" 

59 return None

62def checkout_problem(repo_path: str | Path, *, members: Sequence[str] = ()) -> str | None: 

63 """A single bundle carries one repository's commits: not members, submodules or LFS objects.""" 

64 if members: 

65 return f"it has workspace members ({', '.join(members)}), which a bundle does not carry" 

66 root = Path(repo_path) 

67 if (root / ".gitmodules").exists(): 

68 return "the repo has submodules, which a bundle does not carry" 

69 attrs = root / ".gitattributes" 

70 if attrs.is_file() and "filter=lfs" in attrs.read_text(errors="replace"): 

71 return "the repo uses Git LFS, whose objects a bundle does not carry" 

72 return None

75def filing_problems( 

76 runners: Iterable[str], 

77 *, 

78 workers: registry.WorkersFile, 

79 repos: Sequence[RepoEntry], 

80 in_place: bool, 

81 repo_path: str | Path, 

82 members: Sequence[str] = (), 

83) -> list[str]: 

84 problems = [] 

85 for runner in sorted(set(runners) - {LOCAL}): 

86 if why := refusal(runner, workers=workers, repos=repos): 

87 problems.append(why) 

88 if in_place: 

89 problems.append( 

90 f"runner {runner!r} cannot run an in-place item: it runs in your own checkout " 

91 f"(in-place and runner: {runner} cannot be combined)" 

92 ) 

93 if why := checkout_problem(repo_path, members=members): 

94 problems.append(f"runner {runner!r} cannot take this item: {why}") 

95 return problems

98def remote(runner: str | None) -> str | None: 

99 """What `run_task`'s `runner=` takes: None for `local` and for no placement.""" 

100 return None if runner in (None, LOCAL) else runner

103def node_runner(row, node_id: str) -> str: 

104 """A node's own launches that carry no task (an escalation turn, a review reply).""" 

105 snapshot = store.materialized_chain_of(row) 

106 library = None 

107 if snapshot is not None: 

108 node = next((n for n in snapshot.chain.nodes if n.id == node_id), None) 

109 library = node.runner if node is not None else None 

110 return runner_for(row, library, node_id)
