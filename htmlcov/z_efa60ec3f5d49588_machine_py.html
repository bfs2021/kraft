1"""The machine's side of a connection: `kraft admin worker serve`. 

2 

3Two ways in. Dial-in: this process opens a WebSocket to the server and keeps 

4redialling. `--stdio`: the server ran `ssh host kraft admin worker serve 

5--stdio`, and the protocol is this process's stdin and stdout."""

23def dial_url(server: str) -> str: 

24 """The WebSocket URL for a server's address. A path prefix (a reverse proxy 

25 that serves Kraft under one) is kept.""" 

26 parts = urlsplit(server) 

27 scheme = _SCHEMES.get(parts.scheme) 

28 if scheme is None or not parts.netloc: 

29 raise ValueError(f"server {server!r} must be an http:// or https:// address") 

30 return f"{scheme}://{parts.netloc}{parts.path.rstrip('/')}{CONNECT_PATH}"

41async def dial_in( 

42 server: str, 

43 token: str, 

44 *, 

45 kraft_version: str, 

46 stop: asyncio.Event, 

47 log: Callable[[str], None] = _say, 

48 first_delay: float = 1.0, 

49 max_delay: float = 30.0, 

50) -> None: 

51 """Stay connected to `server` until `stop` is set: redial after a refusal, 

52 a drop or an unreachable server, backing off from `first_delay` to 

53 `max_delay` (reset after a connection that was welcomed). A refused token 

54 or version is retried too: an upgrade or a fresh token fixes it.""" 

55 from websockets.asyncio.client import connect 

56 from websockets.exceptions import WebSocketException 

57 

58 url, delay = dial_url(server), first_delay 

59 

60 async def once() -> bool: 

61 """One connection; True when the server welcomed it.""" 

62 try: 

63 async with connect( 

64 url, max_size=FRAME_LIMIT, ping_interval=None, open_timeout=10 

65 ) as ws: 

66 t = WebSocketTransport(ws.send, ws.recv, ws.close) 

67 welcome = await protocol.machine_handshake( 

68 t, kraft_version=kraft_version, token=token, host=socket.gethostname() 

69 ) 

70 log(f"connected as {welcome['name']}") 

71 try: 

72 await protocol.pump(t, on_frame=_ignore) 

73 except protocol.Closed as exc: 

74 log(f"disconnected: {exc}") 

75 return True 

76 except protocol.Refused as exc: 

77 log(f"refused: {exc}") 

78 except (OSError, TimeoutError, WebSocketException, protocol.Closed) as exc: 

79 log(f"cannot connect: {exc or type(exc).__name__}") 

80 return False 

81 

82 while not stop.is_set(): 

83 attempt = asyncio.ensure_future(once()) 

84 stopper = asyncio.ensure_future(stop.wait()) 

85 await asyncio.wait({attempt, stopper}, return_when=asyncio.FIRST_COMPLETED) 

86 if not attempt.done(): 

87 attempt.cancel() 

88 await asyncio.gather(attempt, return_exceptions=True) 

89 break 

90 stopper.cancel() 

91 delay = first_delay if attempt.result() else min(delay * 2, max_delay) 

92 with contextlib.suppress(TimeoutError): 

93 await asyncio.wait_for(stop.wait(), delay)

96async def run_stdio(*, kraft_version: str, stdin=None, stdout=None) -> None: 

97 """The ssh side: handshake with no token, then keep alive until ssh closes 

98 the pipe. `stdout` is the binary file the protocol is written to, which is 

99 not `sys.stdout`: anything else this process prints must not corrupt it.""" 

100 loop = asyncio.get_running_loop() 

101 reader = asyncio.StreamReader(limit=FRAME_LIMIT) 

102 await loop.connect_read_pipe(lambda: asyncio.StreamReaderProtocol(reader), stdin or sys.stdin) 

103 w_transport, w_protocol = await loop.connect_write_pipe( 

104 asyncio.streams.FlowControlMixin, stdout or sys.stdout 

105 ) 

106 t = StreamTransport(reader, asyncio.StreamWriter(w_transport, w_protocol, reader, loop)) 

107 try: 

108 await protocol.machine_handshake(t, kraft_version=kraft_version, host=socket.gethostname()) 

109 await protocol.pump(t, on_frame=_ignore) 

110 except (protocol.Closed, protocol.Refused): 

111 pass # ssh closed the pipe, or the server said no: either way this process is done 

112 finally: 

113 await t.close()
